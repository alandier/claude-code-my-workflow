\section{Summary and Next Steps}
% ===================================================================

\begin{frame}{Stylized Facts}
\small
\begin{enumerate}\setlength\itemsep{0pt}
\item \textbf{Country dominates P/E} --- 64.6\% of $R^2$ (Shapley), after controlling for size, leverage, earnings growth, industry.

\item \textbf{Forward P/E widens country gaps} --- Removing expected earnings increases dispersion. Country effects reflect \textbf{discount rates}, not earnings.

\item \textbf{Large and persistent} --- $\sigma(\hat\gamma_c) \approx 0.7$ even after R\&D, LTG, macro controls. Japan trades at 0.46$\times$ USA forward P/E. India at 1.22$\times$.

\item \textbf{Real interest rate: null} --- Cross-country real rates do not predict forward P/E ($t = 0.49$). IMF growth forecasts marginal ($t = 1.72$). Consistent with the ``stock-yield disconnect'' (Gormsen--Lazarus 2026).

\item \textbf{Savings: modest predictor} --- Savings/GNI predicts forward P/E ($r = +0.29$), but financial development variables (investor protection, market depth) are null.
\end{enumerate}
\end{frame}

\begin{frame}{Research Questions}
\small
\begin{enumerate}\setlength\itemsep{1pt}
\item \textbf{What drives the forward P/E discount-rate gap?} --- After controlling for firm-level growth, R\&D, and risk proxies, large country effects remain. What are we missing?

\item \textbf{Can the savings channel be formalized?} --- Savings/GNI is the best cross-sectional predictor. A model where savings rates affect discount rates?

\item \textbf{Do low-P/E countries earn higher returns?} --- If so, valuation gaps reflect risk premia. Expected-returns test using Fama--French international factors.

\item \textbf{Pure discounting decomposition} --- Gormsen--Lazarus (2026): can their ``pure discounting'' component explain cross-country forward P/E?

\item \textbf{Why is Japan so cheap?} --- Forward P/E at 0.46$\times$ USA even after analyst growth controls. Korea even lower (0.38$\times$). Cultural, governance, or measurement?
\end{enumerate}
\end{frame}

\begin{frame}{Related Literature}
\footnotesize
\vspace{-0.2cm}
\textbf{Country vs industry effects:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item Heston--Rouwenhorst (1994) --- country dominates returns. We extend to \emph{levels}.
\item Griffin--Karolyi (1998) --- industry effects vary across sectors.
\end{itemize}
\vspace{-0.1cm}
\textbf{Institutions and investor protection:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item La Porta et al.\ (2002) --- investor protection raises Tobin's Q. Weak for forward P/E.
\item Hail--Leuz (2006) --- institutions reduce cost of equity by up to 200 basis points.
\item Stulz (2005) --- ``twin agency problems'' prevent equalization.
\end{itemize}
\vspace{-0.1cm}
\textbf{Savings and asset prices:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item Abel (2003) --- baby boomers' saving and asset prices.
\item Caballero et al.\ (2008) --- global imbalances and asset price inflation.
\end{itemize}
\vspace{-0.1cm}
\textbf{Interest rates and equity valuations:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item Gormsen--Lazarus (2026) --- ``stock-yield disconnect'': only ``pure discounting'' passes through to equities.
\end{itemize}
\end{frame}
