\section{Cross-Country Regressions}

\begin{frame}{Regression Specifications}
\textbf{P/E} (profitable firms, $NI > 0$):
\begin{align*}
\log(P/E)_{i,t} &= \beta_1 \text{Lev}_{i,t} + \beta_2 \log(\text{Assets})_{i,t} + \beta_3 \Delta E_{i,t-1} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\textbf{M/B} (all firms):
\begin{align*}
\log(M/B)_{i,t} &= \beta_1 \text{ROA}_{i,t} + \beta_2 \text{Lev}_{i,t} + \beta_3 \log(\text{Assets})_{i,t} + \beta_4 \Delta E_{i,t-1} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\textbf{ROA}:
\begin{align*}
\text{ROA}_{i,t} &= \beta_1 \text{Lev}_{i,t} + \beta_2 \log(\text{Assets})_{i,t} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\small $\gamma_c$ = country FE, $\delta_j$ = GICS industry FE, $\tau_t$ = year FE.
Standard errors clustered at the country level. Spec 2 replaces separate $\gamma_c + \tau_t$ with $\gamma_{c,t}$ (country$\times$year FE).
\end{frame}

\begin{frame}{P/E vs M/B: Key Regression Differences}
\begin{itemize}
\item \textbf{Leverage}: Insignificant for P/E ($-$0.02) but \emph{dominant} for M/B ($+$0.91***)
\begin{itemize}
\item M/B mechanically increases with leverage: less book equity $\Rightarrow$ higher market/book
\item P/E is unaffected: price and earnings both scale with equity
\end{itemize}

\item \textbf{Earnings growth}: Negative for P/E ($-$0.016), positive for M/B ($+$0.008)
\begin{itemize}
\item P/E: past earnings surge raises denominator $\Rightarrow$ lower P/E
\item M/B: past growth signals quality $\Rightarrow$ higher market premium over book
\end{itemize}

\item \textbf{ROA}: Insignificant in both --- after country and industry FE, current profitability does not predict cross-sectional valuation

\item \textbf{$R^2$}: P/E (0.133) $<$ M/B (0.211) --- M/B partly driven by leverage mechanics
\end{itemize}
\end{frame}

\section{Stylized Facts and Takeaways}
% ===================================================================

\begin{frame}{Are P/E and M/B Results Aligned?}
\small
\textbf{Broadly yes} (country-level correlation = $+$0.85), \textbf{but with key differences}:

\vspace{0.1cm}
\begin{tabular}{@{}p{5.5cm}cc@{}}
\toprule
\textbf{Finding} & \textbf{P/E} & \textbf{M/B} \\
\midrule
Country share of $R^2$ & 64.6\% & 51.6\% \\
Industry share of $R^2$ & 24.3\% & 35.1\% \\
Leverage coefficient & $-$0.02 (ns) & $+$0.91*** \\
Japan effect (avg) & $+$0.01 & $-$0.36 \\
Dispersion trend & $\times$2.7 & $\times$1.1 \\
\bottomrule
\end{tabular}

\vspace{0.1cm}
\textbf{Key insight}: M/B is contaminated by leverage norms and book equity accounting. P/E is a \emph{cleaner} cross-country comparison, revealing even stronger country dependence.
\end{frame}

\begin{frame}{Stylized Facts (1/2): Decomposition}
\begin{enumerate}
\item \textbf{Country effects dominate valuation} --- country explains 52--65\% of $R^2$ (Shapley), more in P/E than M/B. Extends Heston--Rouwenhorst from \emph{returns} to \emph{levels}.

\item \textbf{Country dominates profitability too} --- for ROA, country edges industry (51\% vs 45\%).

\item \textbf{Negative valuation--ROA correlation} ($\rho \approx -0.37$) --- high-valuation countries are \emph{not} high-ROA countries. Consistent across both P/E and M/B.

\item \textbf{M/B is leverage-contaminated} --- leverage coefficient +0.91*** in M/B, insignificant in P/E. Cross-country leverage heterogeneity (std = 0.095) mechanically inflates M/B variation.
\end{enumerate}
\end{frame}

\begin{frame}{Stylized Facts (2/2): Dynamics}
\begin{enumerate}
\setcounter{enumi}{4}
\item \textbf{Japan discount is a measurement artifact} --- M/B effect $-$0.36 (100\% negative years), but P/E effect $\approx 0$. Conservative Japanese book equity accounting drives the M/B gap.

\item \textbf{China premium is real but declining} --- high in both P/E and M/B, but falling since 2015.

\item \textbf{Valuation dispersion is diverging} --- P/E dispersion nearly tripled (0.17 $\to$ 0.46). Despite globalization, country valuations are \emph{not converging}.

\item \textbf{Profitability does not explain valuations} --- ROA coefficient is insignificant in both P/E and M/B regressions after country/industry FE. Discount rates and growth expectations are the key channels.
\end{enumerate}
\end{frame}

\section{Literature}
% ===================================================================

\begin{frame}{Related Literature}
\footnotesize
\textbf{Country vs industry effects:}
\begin{itemize}\setlength\itemsep{0pt}
\item Heston--Rouwenhorst (1994, \textit{JFE}) --- country dominates returns. We extend to \emph{levels}.
\item Griffin--Karolyi (1998) --- industry effects vary across sectors.
\end{itemize}

\textbf{Institutions and investor protection:}
\begin{itemize}\setlength\itemsep{0pt}
\item La Porta et al.\ (2002, \textit{JF}) --- investor protection raises Tobin's Q.
\item Hail--Leuz (2006, \textit{JAR}) --- institutions reduce cost of equity by up to 200 bps.
\item Stulz (2005, \textit{JF}) --- ``twin agency problems'' prevent equalization.
\end{itemize}

\textbf{Market-to-book as a metric:}
\begin{itemize}\setlength\itemsep{0pt}
\item Penman et al.\ (2007, \textit{JAR}) --- decompose M/B into leverage and operating components.
\item Our P/E vs M/B divergence on Japan echoes this concern.
\end{itemize}
\end{frame}

\section{Next Steps}
% ===================================================================

\begin{frame}{Research Questions Raised}
\small
\begin{enumerate}\setlength\itemsep{2pt}
\item \textbf{Why is P/E dispersion diverging?} ($\times$2.7) --- US tech sector? Post-GFC monetary policy? Institutional changes?

\item \textbf{What explains China's P/E premium?} --- Highest P/E despite moderate M/B. Retail investors? Capital controls? Growth expectations?

\item \textbf{Can we decompose the discount rate channel?} --- Country FE mix risk premia, liquidity, governance, investor access

\item \textbf{Why are valuations negatively correlated with ROA?} ($\rho \approx -0.37$) --- Capital allocation inefficiency? Higher risk in profitable countries?

\item \textbf{Japan as natural experiment} --- P/E $\approx 0$ vs M/B $= -0.36$: can we bound the ``accounting noise'' in M/B?
\end{enumerate}
\end{frame}

\begin{frame}{Suggested Next Steps}
\small
\begin{enumerate}\setlength\itemsep{2pt}
\item \textbf{EV/EBITDA} as third metric --- avoids book equity issues (M/B) and profitable-firms restriction (P/E)

\item \textbf{Institutional variables} --- legal origin, anti-self-dealing index, capital controls, sovereign ratings as country-level regressors

\item \textbf{Realized returns} --- do low-valuation countries earn higher returns? Link to Bekaert--Harvey, Fama--French international

\item \textbf{Persistence} --- Markov transition matrices for country valuation quintiles: permanent or mean-reverting?

\item \textbf{Sector decomposition} --- Real Estate shows extreme country effects. Tradable vs non-tradable sectors

\item \textbf{Within-country dispersion} --- does country-level valuation heterogeneity predict aggregate returns?
\end{enumerate}
\end{frame}
