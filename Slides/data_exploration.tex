\section{Summary and Next Steps}
% ===================================================================

\begin{frame}{Stylized Facts}
\small
\begin{enumerate}\setlength\itemsep{3pt}
\item \textbf{Country dominates P/E} --- 65\% of $R^2$ (Shapley), after controlling for size, leverage, earnings growth, industry. Country effects survive firm-level R\&D control ($\sigma$ drops by 0.3\%).

\item \textbf{P/E dispersion is diverging} --- $\times$2.7 since early 2000s.

\item \textbf{Two robust predictors of country P/E effects:}
\begin{itemize}\setlength\itemsep{1pt}
\item \textbf{R\&D intensity} ($t = 5.89$) --- intangible-intensive countries valued higher.
\item \textbf{National savings} ($t = 2.88$) --- more capital seeking equity returns.
\end{itemize}
\vspace{-0.05cm}
$R^2 = 0.45$ with growth control ($n = 65$). Both survive jointly. Growth alone is null.

\item \textbf{Large residual effects persist} --- 74\% unexplained. China premium, Brazil discount survive all controls. Not about growth, institutions, or financial development.
\end{enumerate}
\end{frame}

\begin{frame}{Research Questions}
\small
\begin{enumerate}\setlength\itemsep{4pt}
\item \textbf{Why is P/E dispersion diverging?} ($\times$2.7) --- US tech? Post-GFC monetary policy? Capital flows?

\item \textbf{Can the savings channel be formalized?} --- Savings/GNI predicts P/E and is orthogonal to ROA. A model where savings rates affect discount rates?

\item \textbf{Do low-P/E countries earn higher returns?} --- Bekaert--Harvey, Fama--French international. If so, valuation gaps reflect risk premia.

\item \textbf{What explains China's P/E premium?} --- Highest P/E among large countries. Retail investors? Capital controls? Savings rate (46\% of GNI)?
\end{enumerate}
\end{frame}

\begin{frame}{Related Literature}
\footnotesize
\textbf{Country vs industry effects:}
\begin{itemize}\setlength\itemsep{0pt}
\item Heston--Rouwenhorst (1994, \textit{JFE}) --- country dominates returns. We extend to \emph{levels}.
\item Griffin--Karolyi (1998, \textit{JFE}) --- industry effects vary across sectors.
\end{itemize}

\textbf{Institutions and investor protection:}
\begin{itemize}\setlength\itemsep{0pt}
\item La Porta et al.\ (2002, \textit{JF}) --- investor protection raises Tobin's Q. Our results: weak for P/E.
\item Hail--Leuz (2006, \textit{JAR}) --- institutions reduce cost of equity by up to 200 bps.
\item Stulz (2005, \textit{JF}) --- ``twin agency problems'' prevent equalization.
\end{itemize}

\textbf{Savings and asset prices:}
\begin{itemize}\setlength\itemsep{0pt}
\item Abel (2003, \textit{Brookings}) --- effect of baby boomers' saving on asset prices.
\item Caballero et al.\ (2008, \textit{JME}) --- global imbalances and asset price inflation.
\end{itemize}
\end{frame}

\begin{frame}{Market Cap Construction (Detail)}
\footnotesize
\textbf{US firms}: \texttt{mkvalt} (pre-computed market value in USD millions). Available from 1986.

\vspace{0.1cm}
\textbf{Non-US firms} (Compustat Global): no pre-computed market value.
\begin{itemize}\setlength\itemsep{0pt}
\item Compute: Market Cap = \texttt{cshoi} (shares outstanding) $\times$ \texttt{prccm} (month-end price)
\item \textbf{Pitfall}: firms have multiple securities across exchanges and currencies

\vspace{0.05cm}
\emph{Example --- SAP SE (gvkey=103487) in Dec 2022:}
\begin{tabular}{@{}llr@{}}
iid=02W & EUR & 96.39 $\leftarrow$ \textcolor{accent}{correct} \\
iid=05W & HUF & 38{,}620 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
iid=06W & MXN & 2{,}009 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
\end{tabular}

\item \textbf{Solution}: use \texttt{prirow} field (primary security designation)
\end{itemize}

\vspace{0.05cm}
All data in local currency (millions). Cross-country \emph{levels} absorbed by country FE.
\end{frame}
