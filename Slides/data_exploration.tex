\section{Summary and Next Steps}
% ===================================================================

\begin{frame}{Stylized Facts}
\small
\begin{enumerate}\setlength\itemsep{4pt}
\item \textbf{Country dominates P/E} --- 65\% of $R^2$ (Shapley). Extends Heston--Rouwenhorst (1994) from returns to valuation levels.

\item \textbf{P/E dispersion is diverging} --- $\times$2.7 since early 2000s, despite financial globalization.

\item \textbf{National savings rate predicts P/E} ($r = +0.34$, $t = 3.01$). More savings $\Rightarrow$ more capital chasing equities $\Rightarrow$ higher multiples. The only significant macro predictor.

\item \textbf{Standard institutional variables are null} --- La Porta, financial development, GDP per capita: all insignificant.

\item \textbf{High-P/E countries are less profitable} ($r = -0.41$). Consistent with a discount rate channel rather than a cash flow story.
\end{enumerate}
\end{frame}

\begin{frame}{Research Questions}
\small
\begin{enumerate}\setlength\itemsep{4pt}
\item \textbf{Why is P/E dispersion diverging?} ($\times$2.7) --- US tech? Post-GFC monetary policy? Capital flows?

\item \textbf{Can the savings channel be formalized?} --- Savings/GNI predicts P/E and is orthogonal to ROA. A model where savings rates affect discount rates?

\item \textbf{Do low-P/E countries earn higher returns?} --- Bekaert--Harvey, Fama--French international. If so, valuation gaps reflect risk premia.

\item \textbf{What explains China's P/E premium?} --- Highest P/E among large countries. Retail investors? Capital controls? Savings rate (46\% of GNI)?
\end{enumerate}
\end{frame}

\begin{frame}{Related Literature}
\footnotesize
\textbf{Country vs industry effects:}
\begin{itemize}\setlength\itemsep{0pt}
\item Heston--Rouwenhorst (1994, \textit{JFE}) --- country dominates returns. We extend to \emph{levels}.
\item Griffin--Karolyi (1998, \textit{JFE}) --- industry effects vary across sectors.
\end{itemize}

\textbf{Institutions and investor protection:}
\begin{itemize}\setlength\itemsep{0pt}
\item La Porta et al.\ (2002, \textit{JF}) --- investor protection raises Tobin's Q. Our results: weak for P/E.
\item Hail--Leuz (2006, \textit{JAR}) --- institutions reduce cost of equity by up to 200 bps.
\item Stulz (2005, \textit{JF}) --- ``twin agency problems'' prevent equalization.
\end{itemize}

\textbf{Savings and asset prices:}
\begin{itemize}\setlength\itemsep{0pt}
\item Abel (2003, \textit{Brookings}) --- effect of baby boomers' saving on asset prices.
\item Caballero et al.\ (2008, \textit{JME}) --- global imbalances and asset price inflation.
\end{itemize}
\end{frame}

\begin{frame}{P/E Regression Specification}
\small

\vspace{-0.2cm}
$$\log(P/E)_{i,t} = \beta_1 \text{Lev} + \beta_2 \log(\text{Assets}) + \beta_3 \Delta E_{t-1} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}$$

\vspace{-0.1cm}
\begin{tabular}{@{}lcc@{}}
\toprule
& Spec 1 & Spec 2 \\
\midrule
Leverage & $-$0.017 (0.102) & $-$0.013 (0.100) \\
$\log(\text{At})$ & $-$0.076** (0.030) & $-$0.078** (0.031) \\
Lag $\Delta E$ & $-$0.016*** (0.003) & $-$0.016*** (0.003) \\
\midrule
FE & Country + Ind + Year & Country$\times$Year + Ind \\
N / $R^2$ & 428K / 0.142 & 428K / 0.163 \\
\bottomrule
\end{tabular}

\vspace{0.1cm}

\begin{frame}{Profitability Puzzle: Robustness Details}
\footnotesize
\begin{tabular}{@{}lcc@{}}
\toprule
\textbf{Specification} & \textbf{$r$(ROA, P/E)} & \textbf{Note} \\
\midrule
Raw country averages & $-$0.02 & No controls \\
Industry$\times$year demeaned & $+$0.11 & Industry removed \\
\addlinespace
Regression \textbf{without} industry FE & $-$0.46 & Size + leverage + earn.\ growth \\
Regression \textbf{with} industry FE & $-$0.41 & Main specification \\
Using lagged ROA$_{t-1}$ & $-$0.37 & Breaks mechanical link \\
\addlinespace
\textbf{With USD size controls:} \\
Both P/E and ROA control for size & $-$0.64 & Strongest \\
No size controls on either side & $-$0.02 & Disappears \\
\bottomrule
\end{tabular}

\vspace{0.1cm}
\footnotesize The negative correlation is a \textbf{Simpson's paradox}: within each size bucket, high-ROA countries have lower P/E. Without size controls, this is masked. Effect is stronger with USD-converted assets ($-$0.64 vs $-$0.51 local), ruling out a currency artifact.
\end{frame}

\begin{frame}{Market Cap Construction (Detail)}
\footnotesize
\textbf{US firms}: \texttt{mkvalt} (pre-computed market value in USD millions). Available from 1986.

\vspace{0.1cm}
\textbf{Non-US firms} (Compustat Global): no pre-computed market value.
\begin{itemize}\setlength\itemsep{0pt}
\item Compute: Market Cap = \texttt{cshoi} (shares outstanding) $\times$ \texttt{prccm} (month-end price)
\item \textbf{Pitfall}: firms have multiple securities across exchanges and currencies

\vspace{0.05cm}
\emph{Example --- SAP SE (gvkey=103487) in Dec 2022:}
\begin{tabular}{@{}llr@{}}
iid=02W & EUR & 96.39 $\leftarrow$ \textcolor{accent}{correct} \\
iid=05W & HUF & 38{,}620 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
iid=06W & MXN & 2{,}009 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
\end{tabular}

\item \textbf{Solution}: use \texttt{prirow} field (primary security designation)
\end{itemize}

\vspace{0.05cm}
All data in local currency (millions). Cross-country \emph{levels} absorbed by country FE.
\end{frame}
