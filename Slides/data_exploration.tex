\begin{frame}{Variable Definitions}
\small
\textbf{Dependent variables:}
\vspace{-0.2cm}
\begin{align*}
\log(P/E)_{i,t} &= \log\!\left(\frac{\text{Market Cap}_{i,t}}{\text{Net Income}_{i,t}}\right), \quad NI > 0 \\[-2pt]
\log(M/B)_{i,t} &= \log\!\left(\frac{\text{Market Cap}_{i,t}}{\text{Book Equity}_{i,t}}\right)
\end{align*}

\vspace{-0.35cm}
\begin{tabular}{@{}p{2.8cm}p{10cm}@{}}
Market Cap & US: \texttt{mkvalt}. Global: \texttt{cshoi} $\times$ \texttt{prccm} (primary security) \\
Book Equity & \texttt{ceq} (common equity). \emph{Does not include debt.} \\
Net Income & \texttt{ib} (income before extraordinary items) \\
Leverage & $(\texttt{at} - \texttt{ceq}) / \texttt{at}$ = non-equity share of assets \\
ROA & \texttt{ib} / \texttt{at}. \quad Industry: GICS Industry Group ($\sim$25 categories)
\end{tabular}

\vspace{0.05cm}
\footnotesize All ratios winsorized at 1st/99th percentiles. All data in local currency (millions).
\end{frame}

\begin{frame}{Market Cap Construction}
\footnotesize
\textbf{US firms}: \texttt{mkvalt} (pre-computed market value in USD millions). Available from 1986.

\vspace{0.1cm}
\textbf{Non-US firms} (Compustat Global): no pre-computed market value.
\begin{itemize}\setlength\itemsep{0pt}
\item Compute: Market Cap = \texttt{cshoi} (shares outstanding) $\times$ \texttt{prccm} (month-end price)
\item \textbf{Pitfall}: firms have multiple securities across exchanges and currencies

\vspace{0.05cm}
\emph{Example --- SAP SE (gvkey=103487) in Dec 2022:}
\begin{tabular}{@{}llr@{}}
iid=02W & EUR & 96.39 $\leftarrow$ \textcolor{accent}{correct} \\
iid=05W & HUF & 38{,}620 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
iid=06W & MXN & 2{,}009 $\leftarrow$ \textcolor{alert}{wrong currency!} \\
\end{tabular}

\item \textbf{Solution}: use \texttt{prirow} field (primary security designation) to select the correct iid for each firm
\end{itemize}

\vspace{0.05cm}
All data in local currency (millions). Cross-country \emph{levels} absorbed by country FE.
\end{frame}

\section{Cross-Country Regressions}

\begin{frame}{Regression Specifications}
\small
\textbf{P/E} (profitable firms, $NI > 0$):
\vspace{-0.2cm}
\begin{align*}
\log(P/E)_{i,t} &= \beta_1 \text{Lev} + \beta_2 \log(\text{Assets}) + \beta_3 \Delta E_{t-1} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\vspace{-0.4cm}
\textbf{M/B} (all firms):
\vspace{-0.2cm}
\begin{align*}
\log(M/B)_{i,t} &= \beta_1 \text{ROA} + \beta_2 \text{Lev} + \beta_3 \log(\text{Assets}) + \beta_4 \Delta E_{t-1} + \text{FE} + \varepsilon_{i,t}
\end{align*}

\vspace{-0.4cm}
\textbf{ROA}:
\vspace{-0.2cm}
\begin{align*}
\text{ROA}_{i,t} &= \beta_1 \text{Lev} + \beta_2 \log(\text{Assets}) + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\vspace{-0.3cm}
\footnotesize FE = $\gamma_c$ (country) + $\delta_j$ (GICS industry) + $\tau_t$ (year).
SE clustered at country level. Spec 2: $\gamma_{c,t}$ (country$\times$year) replaces $\gamma_c + \tau_t$.
\end{frame}

\end{frame}

\begin{frame}{P/E vs M/B: Key Regression Differences}
\begin{itemize}
\item \textbf{Leverage}: Insignificant for P/E ($-$0.02) but dominant for M/B ($+$0.90***)
\begin{itemize}
\item M/B: less book equity $\Rightarrow$ higher market/book (mechanical)
\item P/E: price and earnings both scale with equity $\Rightarrow$ no effect
\end{itemize}

\item \textbf{Earnings growth}: Negative for P/E ($-$0.016***), positive for M/B ($+$0.008***)
\begin{itemize}
\item P/E: past earnings surge raises denominator $\Rightarrow$ lower P/E
\item M/B: past growth signals quality $\Rightarrow$ higher market premium
\end{itemize}

\item \textbf{ROA}: Not included in P/E spec; insignificant in M/B after controls --- current profitability does not predict cross-sectional valuation

\item \textbf{$R^2$}: P/E (0.142) $<$ M/B (0.219) --- M/B partly driven by leverage mechanics
\end{itemize}
\end{frame}

\begin{frame}{The Correlation Is Not Mechanical (Lagged ROA Test)}
\vspace{-0.2cm}
\centering
\includegraphics[height=0.52\textheight]{scatter_lag_roa_vs_pe.pdf}

\vspace{-0.05cm}
\footnotesize
\textbf{Concern}: NI is in both ROA (numerator) and P/E (denominator) --- a temporary earnings shock mechanically raises ROA and lowers P/E.

\textbf{Test}: replace ROA$_t$ with ROA$_{t-1}$. Last year's NI is not in this year's P/E.

\textbf{Result}: correlation barely changes ($-$0.41 $\to$ $-$0.37). Country ROA effects are extremely persistent ($r = 0.99$), so the link reflects \emph{permanent} profitability differences, not transitory earnings.
\end{frame}

\section{Stylized Facts and Takeaways}
% ===================================================================

\begin{frame}{Are P/E and M/B Results Aligned?}
\small
\textbf{Broadly yes} (country-level correlation = $+$0.85), \textbf{but with key differences}:

\vspace{0.15cm}
\begin{tabular}{@{}p{5cm}cc@{}}
\toprule
\textbf{Finding} & \textbf{P/E} & \textbf{M/B} \\
\midrule
Country share of $R^2$ & 64.6\% & 50.8\% \\
Industry share of $R^2$ & 24.8\% & 35.8\% \\
Leverage coefficient & $-$0.02 (ns) & $+$0.90*** \\
Japan effect (avg) & $+$0.03 & $-$0.35 \\
Dispersion trend & $\times$2.7 & $\times$1.1 \\
\bottomrule
\end{tabular}

\vspace{0.15cm}
\textbf{Key insight}: M/B is contaminated by leverage norms and book equity accounting (IFRS vs domestic GAAP). P/E reveals even \emph{stronger} country dependence.
\end{frame}

\begin{frame}{Stylized Facts (1/2): Cross-Sectional Structure}
\small
\begin{enumerate}\setlength\itemsep{3pt}
\item \textbf{Country effects dominate valuation} --- country explains 51--65\% of $R^2$ (Shapley), more in P/E than M/B. Extends Heston--Rouwenhorst (1994) from \emph{returns} to \emph{levels}.

\item \textbf{Country dominates profitability too} --- for ROA, country edges industry (51\% vs 45\%).

\item \textbf{Negative valuation--ROA correlation across countries} ($\rho \approx -0.4$) --- \emph{not} explained by financial development (La Porta), GDP per capita, or firm size (in USD). Within countries, P/E--ROA flips positive ($+$0.80). The cross-country puzzle remains open.

\item \textbf{M/B is leverage-contaminated} --- leverage coefficient $+$0.90*** in M/B, insignificant in P/E. Cross-country leverage heterogeneity (std = 0.095) mechanically inflates M/B variation.
\end{enumerate}
\end{frame}

\begin{frame}{Stylized Facts (2/2): Dynamics}
\small
\begin{enumerate}\setlength\itemsep{3pt}
\setcounter{enumi}{4}
\item \textbf{Japan discount is mostly a measurement artifact} --- M/B effect $-$0.35 (100\% negative years), but P/E effect $\approx 0$. Conservative Japanese book equity drives the M/B gap.

\item \textbf{China premium is real but declining} --- high in both P/E and M/B, but falling since 2015.

\item \textbf{Valuation dispersion is diverging} --- P/E dispersion nearly tripled. Despite financial globalization, country valuations are \emph{not converging}.

\item \textbf{US VW premium is enormous} --- driven by large cap: $+$54\% P/E and $+$258\% M/B vs Japan (VW). The US premium is much smaller in EW, suggesting it is concentrated in large tech firms.
\end{enumerate}
\end{frame}

\begin{frame}{Related Literature}
\footnotesize
\textbf{Country vs industry effects:}
\begin{itemize}\setlength\itemsep{0pt}
\item Heston--Rouwenhorst (1994, \textit{JFE}) --- country dominates returns. We extend to \emph{levels}.
\item Griffin--Karolyi (1998, \textit{JFE}) --- industry effects vary across sectors.
\end{itemize}

\textbf{Institutions and investor protection:}
\begin{itemize}\setlength\itemsep{0pt}
\item La Porta et al.\ (2002, \textit{JF}) --- investor protection raises Tobin's Q.
\item Hail--Leuz (2006, \textit{JAR}) --- institutions reduce cost of equity by up to 200 bps.
\item Stulz (2005, \textit{JF}) --- ``twin agency problems'' prevent equalization.
\end{itemize}

\textbf{Market-to-book and accounting:}
\begin{itemize}\setlength\itemsep{0pt}
\item Penman et al.\ (2007, \textit{JAR}) --- decompose M/B into leverage and operating components.
\item Our P/E vs M/B divergence on Japan echoes this concern about M/B.
\end{itemize}
\end{frame}

\section{Research Agenda}
% ===================================================================

\begin{frame}{Research Questions}
\small
\begin{enumerate}\setlength\itemsep{3pt}
\item \textbf{Why is P/E dispersion diverging?} ($\times$2.7) --- US tech sector? Post-GFC monetary policy? Institutional changes?

\item \textbf{What explains China's P/E premium?} --- Highest P/E despite moderate M/B. Retail investors? Capital controls? Growth expectations?

\item \textbf{Can we decompose the country effect?} --- Discount rates, growth expectations, investor protection, capital controls, accounting norms

\item \textbf{Why are valuations negatively correlated with ROA?} ($\rho \approx -0.4$) --- Not explained by financial development, investor protection, GDP per capita, or firm size (in USD). Local-currency size appeared to work but was a currency artifact. A genuine open puzzle.

\item \textbf{Japan as natural experiment} --- P/E $\approx 0$ vs M/B $= -0.35$: can we bound the ``accounting noise'' in M/B?
\end{enumerate}
\end{frame}

\begin{frame}{Suggested Next Steps}
\small
\begin{enumerate}\setlength\itemsep{2pt}
\item \textbf{EV/EBITDA} as third metric --- avoids book equity issues (M/B) and profitable-firms restriction (P/E)

\item \textbf{Institutional variables} --- La Porta variables surprisingly weak; try capital controls, FDI openness, expected growth proxies, risk measures

\item \textbf{Realized returns} --- do low-valuation countries earn higher returns? (Bekaert--Harvey, Fama--French international)

\item \textbf{Persistence} --- Markov transition matrices for country valuation quintiles

\item \textbf{Sector decomposition} --- tradable vs non-tradable; Real Estate shows extreme country effects

\item \textbf{Within-country dispersion} --- does valuation heterogeneity predict aggregate returns?
\end{enumerate}
\end{frame}
