\section{Motivation}
% ===================================================================

\begin{frame}{The Question}
\large

Price-earnings ratios vary dramatically across countries.

\vspace{0.3cm}

\textbf{How much of this is firm composition vs.\ a pure country effect?}

\vspace{0.3cm}
\normalsize
Gordon model: \quad $P/E = \dfrac{1}{r - g}$

\vspace{0.15cm}
\begin{itemize}\setlength\itemsep{4pt}
\item Forward P/E uses analyst EPS forecasts $\Rightarrow$ expected growth is controlled
\item Residual country effects must reflect \textbf{discount rates} or \textbf{model misspecification}
\item We also measure expected returns directly (analyst price targets)
\end{itemize}

\vspace{0.2cm}
\textbf{Approach:} We use 200K firm-years across 100+ countries to decompose cross-country P/E differences into discount rate and growth components.
\end{frame}

\begin{frame}{The Puzzle in One Picture}
\centering
\includegraphics[height=0.62\textheight]{country_effects_forward_baseline.pdf}

\vspace{0.05cm}
\scriptsize Market-cap-weighted regional aggregates. Year-by-year regressions controlling for firm size, leverage, R\&D, growth, and 2-digit SIC industry. USA $= 1.0$.

\vspace{0.05cm}
\normalsize
\textbf{Europe has traded at a 10--30\% P/E discount to the US for 25 years.}
\end{frame}

\section{Summary}
% ===================================================================

\begin{frame}{Three Facts About Cross-Country P/E}
\vspace{0.1cm}
\begin{enumerate}\setlength\itemsep{8pt}

\item[\textcolor{accent}{\Large 1.}] {\large \textbf{Country effects are as large as industry effects.}}\\[2pt]
\footnotesize After controlling for firm size, leverage, R\&D, and growth with Country$\times$Year + SIC2$\times$Year FE: $\sigma(\hat\gamma_c) = 0.34$ in $\log(P/E)$, equivalent to a \textcolor{alert}{\textbf{40\% P/E premium}} per standard deviation. Europe trades at 0.7--0.9$\times$ USA for 25 years. The gap is \textbf{widening}.

\item[\textcolor{accent}{\Large 2.}] {\large \textbf{Discount rates go in the right direction but explain little.}}\\[2pt]
\footnotesize P/E and expected return FE correlate strongly ($r = -0.57$), but ER absorbs only \textcolor{alert}{\textbf{14\%}} of $\sigma(\hat\gamma_c)$. Growth expectations absorb 6\%. Together, 16\%.

\item[\textcolor{accent}{\Large 3.}] {\large \textbf{Financial development predicts the gap. Governance does not.}}\\[2pt]
\footnotesize Mkt cap/GDP ($r = 0.48$), credit/GDP ($r = 0.39$), and log GDP/cap ($r = 0.33$) predict higher P/E. Anti-self-dealing is null ($t = 1.7$), contrary to La Porta et al.\ (2002).

\end{enumerate}

\vspace{0.15cm}
\centering
\colorbox{alert!15}{\footnotesize \textbf{84\% of cross-country P/E variation remains unexplained.}}
\end{frame}

\begin{frame}{Research Questions}
\small
\begin{enumerate}\setlength\itemsep{2pt}
\item \textbf{What is the missing factor?}\\
{\footnotesize Persistent country effects unexplained by firm characteristics, growth, or risk. Behavioral? Cultural? Tax? Liquidity?}

\item \textbf{Do analyst expected returns predict realized returns?}\\
{\footnotesize High analyst ER $\Rightarrow$ high realized returns = risk-based. Otherwise = behavioral.}

\item \textbf{Does the ``Korea discount'' reflect governance?}\\
{\footnotesize Korea: P/E $0.61\times$, ER $+$33pp. Classical governance story, but anti-self-dealing index is null ($t = 1.7$).}
\end{enumerate}
\end{frame}

\begin{frame}{Related Literature}
\footnotesize
\vspace{-0.2cm}
\textbf{Country vs industry effects:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item Heston--Rouwenhorst (1994) --- country dominates returns. We extend to \emph{valuation levels}.
\item Griffin--Karolyi (1998) --- industry effects vary across sectors.
\end{itemize}
\vspace{-0.1cm}
\textbf{Institutions and cost of capital:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item La Porta et al.\ (2002) --- investor protection raises Tobin's Q. We find: \textbf{null for forward P/E}.
\item Djankov et al.\ (2008) --- anti-self-dealing index. \textbf{Null} as P/E predictor ($t = 1.7$).
\item Hail--Leuz (2006) --- institutions reduce cost of equity. Our expected return evidence is mixed.
\item Stulz (2005) --- ``twin agency problems'' prevent equalization.
\end{itemize}
\vspace{-0.1cm}
\textbf{Analyst forecasts and pricing:}
\vspace{-0.1cm}
\begin{itemize}\setlength{\itemsep}{-2pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
\item Da--Warachka (2011) --- analyst forecasts and the cross-section of expected returns.
\item Bradshaw et al.\ (2012) --- analysts' forecasts reflect risk differentials.
\end{itemize}
\end{frame}
