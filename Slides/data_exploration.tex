\section{Cross-Country Regressions}

\begin{frame}{Regression Specifications}
\textbf{Market-to-Book:}
\begin{align*}
\log(M/B)_{i,t} &= \alpha + \beta_1 \text{ROA}_{i,t} + \beta_2 \log(\text{Assets})_{i,t} + \beta_3 \Delta E_{i,t-1} \\
&\quad + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t} \tag{Spec 1}
\end{align*}
\begin{align*}
\log(M/B)_{i,t} &= \alpha + \beta_1 \text{ROA}_{i,t} + \beta_2 \log(\text{Assets})_{i,t} + \beta_3 \Delta E_{i,t-1} \\
&\quad + \delta_j + \gamma_{c,t} + \varepsilon_{i,t} \tag{Spec 2}
\end{align*}

\textbf{ROA:}
\begin{align*}
\text{ROA}_{i,t} &= \alpha + \beta_1 \log(\text{Assets})_{i,t} + \gamma_c + \delta_j + \tau_t + \varepsilon_{i,t}
\end{align*}

\small $\gamma_c$ = country FE, $\delta_j$ = industry FE (SIC-2), $\tau_t$ = year FE, $\gamma_{c,t}$ = country $\times$ year FE.\\
Standard errors clustered at the country level.
\end{frame}

\begin{frame}{Key Takeaway: ROA Does Not Explain M/B}
\begin{itemize}
\item ROA coefficient is \textbf{insignificant} in both specifications
\item After country, industry, and year controls, current profitability does not predict M/B
\item Only \textbf{lag earnings growth} is significant (positive, as expected)
\item Size effect negative but insignificant with clustered SEs
\item \textbf{Interpretation}: cross-country M/B differences reflect \emph{discount rates} and \emph{growth expectations}, not current profitability
\end{itemize}

\vspace{0.2cm}
\textbf{$R^2$ comparison:} Spec 1 = 0.185, Spec 2 = 0.211 --- country$\times$year interactions gain 2.6 pp, suggesting time-varying country effects matter.
\end{frame}

\section{Stylized Facts}
% ===================================================================

\begin{frame}{Stylized Facts (1/2): Decomposition}
\begin{enumerate}
\item \textbf{Country effects dominate valuation} --- country explains 50\% of M/B $R^2$ (Shapley), vs 38\% for industry. Echoes the Heston--Rouwenhorst debate, but for \emph{levels} rather than returns.

\item \textbf{Country dominates profitability too} --- for ROA, country edges industry (51\% vs 43\%). Both valuation and profitability are primarily location-driven.

\item \textbf{Negative M/B--ROA correlation across countries} ($\rho = -0.26$) --- high-M/B countries are \emph{not} high-ROA countries. Discount rates and growth expectations, not current profitability, drive the cross-country valuation map.
\end{enumerate}
\end{frame}

\begin{frame}{Stylized Facts (2/2): Dynamics}
\begin{enumerate}
\setcounter{enumi}{3}
\item \textbf{Japan discount is persistent} --- M/B effect negative in \textbf{100\%} of years (avg $-$0.54 vs USA), with no sign of convergence.

\item \textbf{China premium collapsed post-2015} --- from $+0.8$ to near zero, consistent with market correction and deleveraging.

\item \textbf{Cross-country valuation dispersion is increasing} --- despite financial globalization, countries are \emph{diverging} in M/B (std dev: 0.40 $\to$ 0.47).
\end{enumerate}
\end{frame}

\section{Literature}
% ===================================================================

\begin{frame}{Related Literature: Decomposition and Institutions}
\small
\textbf{Country vs industry effects:}
\begin{itemize}
\item Heston--Rouwenhorst (1994, \textit{JFE}) --- country effects dominate for returns. We find the same for \emph{valuation levels}.
\item Griffin--Karolyi (1998, \textit{JFE}) --- industry effects matter more in traded-goods sectors.
\item Brooks--Del Negro (2004) --- country effects declined post-1999 for returns. Opposite for M/B.
\end{itemize}

\vspace{0.1cm}
\textbf{Legal origin and investor protection:}
\begin{itemize}
\item La Porta et al.\ (2002, \textit{JF}) --- investor protection raises Tobin's Q.
\item Djankov et al.\ (2008, \textit{JFE}) --- anti-self-dealing index predicts valuations.
\item Doidge et al.\ (2007, \textit{JFE}) --- countries matter more than firms for governance.
\end{itemize}
\end{frame}

\begin{frame}{Related Literature: Globalization and Dynamics}
\small
\textbf{Limits of convergence:}
\begin{itemize}
\item Stulz (2005, \textit{JF}) --- ``twin agency problems'' prevent equalization.
\item Bekaert et al.\ (2011, \textit{RFS}) --- segmentation decreased but persists.
\item Doidge et al.\ (2004, \textit{JFE}) --- cross-listed firms gain $\sim$17\% Q premium.
\end{itemize}

\vspace{0.1cm}
\textbf{Country-specific puzzles and fundamentals:}
\begin{itemize}
\item French--Poterba (1991, \textit{JFE}) --- Japan P/E premium; persistent discount since.
\item Fama--French (1998, \textit{JF}) --- value premium exists internationally.
\item Rajan--Zingales (1998, \textit{AER}) --- financial development raises valuations.
\end{itemize}
\end{frame}

\begin{frame}{Our Contribution}
\begin{itemize}
\item First panel decomposition of \emph{valuation levels} (not returns) into country, industry, and time using Shapley $R^2$

\item Largest cross-country valuation panel: 81 countries, 1986--2025, 749K firm-years

\item Document \textbf{divergence} in M/B across countries, challenging the globalization-implies-equalization view

\item Country M/B effects \textbf{not} explained by ROA ($\rho = -0.26$) --- points to discount-rate and institutional channels
\end{itemize}
\end{frame}
